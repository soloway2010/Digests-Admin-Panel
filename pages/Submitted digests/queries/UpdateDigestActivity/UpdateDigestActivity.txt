UPDATE public.digest_activities
SET name = '{{this.params.name}}'
WHERE id = {{this.params.id}}
AND digest_id IN (
  SELECT id FROM public.digests
  WHERE {{
    ["p.solovev@sdgroup.ai"].map((e) => e.toLowerCase()).includes((appsmith.user.email || "").toLowerCase())
      ? "TRUE"
      : "user_name ILIKE '%" + (appsmith.user.email || "").split("@")[0].replace(/'/g, "''") + "%'"
  }}
);

UPDATE public.user_activities
SET name = '{{this.params.name}}'
WHERE id = {{this.params.user_activity_id}}
AND {{
  ["p.solovev@sdgroup.ai"].map((e) => e.toLowerCase()).includes((appsmith.user.email || "").toLowerCase())
    ? "TRUE"
    : "user_id IN (SELECT DISTINCT user_id FROM public.digests WHERE user_name ILIKE '%" + (appsmith.user.email || "").split("@")[0].replace(/'/g, "''") + "%')"
}};
